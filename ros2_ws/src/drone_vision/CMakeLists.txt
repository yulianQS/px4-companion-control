cmake_minimum_required(VERSION 3.16)
project(drone_vision LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs aruco calib3d)
find_package(Threads REQUIRED)
find_library(AW_ISP_LIBRARY AWIspApi REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Target2D.msg" DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_library(vision_algorithms STATIC src/detector.cpp src/pose.cpp)
target_include_directories(vision_algorithms PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_algorithms PUBLIC ${OpenCV_LIBS})
add_executable(camera_node src/camera_node.cpp src/mipi_camera.cpp)
target_include_directories(camera_node PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(camera_node rclcpp sensor_msgs std_msgs)
target_link_libraries(camera_node vision_algorithms ${OpenCV_LIBS} ${AW_ISP_LIBRARY} Threads::Threads)
add_executable(tag_detector_node src/tag_detector_node.cpp)
ament_target_dependencies(tag_detector_node rclcpp sensor_msgs std_msgs)
target_link_libraries(tag_detector_node vision_algorithms "${typesupport}" Threads::Threads)
add_executable(follow_preview_node src/follow_preview_node.cpp)
target_include_directories(follow_preview_node PRIVATE include "${CMAKE_CURRENT_SOURCE_DIR}/../../../cpp/vendor")
ament_target_dependencies(follow_preview_node rclcpp sensor_msgs geometry_msgs std_msgs)
target_link_libraries(follow_preview_node "${typesupport}")
if(BUILD_TESTING)
  enable_testing()
  add_executable(vision_algorithms_test test/algorithms_test.cpp)
  target_link_libraries(vision_algorithms_test vision_algorithms)
  add_test(NAME vision_algorithms_test COMMAND vision_algorithms_test)
  add_executable(vision_pose_test test/pose_test.cpp)
  target_link_libraries(vision_pose_test vision_algorithms)
  add_test(NAME vision_pose_test COMMAND vision_pose_test)
  add_executable(vision_test_image_source test/image_source.cpp)
  ament_target_dependencies(vision_test_image_source rclcpp sensor_msgs std_msgs)
endif()
install(TARGETS camera_node tag_detector_node follow_preview_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../../config/vision.yaml" DESTINATION share/${PROJECT_NAME}/config)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
